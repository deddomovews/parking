import cv2
import numpy as np
import os
import urllib.request
from pathlib import Path


def download_yolo_files():
    """Скачивает необходимые файлы для YOLO"""
    files = {
        'yolov4.weights': 'https://github.com/AlexeyAB/darknet/releases/download/darknet_yolo_v3_optimal/yolov4.weights',
        'yolov4.cfg': 'https://raw.githubusercontent.com/AlexeyAB/darknet/master/cfg/yolov4.cfg',
        'coco.names': 'https://raw.githubusercontent.com/pjreddie/darknet/master/data/coco.names'
    }

    for filename, url in files.items():
        if not os.path.exists(filename):
            print(f"Скачивание {filename}...")
            try:
                urllib.request.urlretrieve(url, filename)
                print(f"Файл {filename} скачан!")
            except Exception as e:
                print(f"Ошибка скачивания {filename}: {e}")
                return False
        else:
            print(f"Файл {filename} уже существует")
    return True


def detect_cars_yolo():
    # Скачиваем файлы YOLO
    if not download_yolo_files():
        print("Не удалось скачать файлы YOLO. Проверьте подключение к интернету.")
        return

    # Загружаем YOLO
    print("Загрузка модели YOLOv4...")
    try:
        net = cv2.dnn.readNet("yolov4.weights", "yolov4.cfg")
        print("Модель YOLOv4 загружена!")
    except Exception as e:
        print(f"Ошибка загрузки модели: {e}")
        return

    # Определяем выходные слои
    layer_names = net.getLayerNames()

    # Для разных версий OpenCV может быть разный формат
    try:
        output_layers = [layer_names[i - 1] for i in net.getUnconnectedOutLayers()]
    except:
        output_layers = [layer_names[i[0] - 1] for i in net.getUnconnectedOutLayers()]

    # Загружаем классы COCO
    try:
        with open("coco.names", "r") as f:
            classes = [line.strip() for line in f.readlines()]
        print(f"Загружено классов: {len(classes)}")
    except Exception as e:
        print(f"Ошибка загрузки файла классов: {e}")
        return

    # Пробуем разные источники видео
    VIDEO_SOURCES = [
        "test_images/parking.mp4",
        0,  # веб-камера
        "parking.mp4",
        "video.mp4"
    ]

    cap = None
    video_source = None
    for source in VIDEO_SOURCES:
        print(f"Попытка открыть видео источник: {source}")
        cap = cv2.VideoCapture(source)
        if cap.isOpened():
            video_source = source
            print(f"Успешно открыт источник: {source}")
            break
        else:
            print(f"Не удалось открыть: {source}")

    if cap is None or not cap.isOpened():
        print("Ошибка: Не удалось открыть ни один видео источник!")
        print("Создайте папку 'test_images' и поместите туда видео 'parking.mp4'")
        print("Или подключите веб-камеру")
        return

    # Создаем папку для тестового видео если используем веб-камеру
    if video_source == 0:
        os.makedirs("test_images", exist_ok=True)

    # Настройки для оптимизации производительности
    net.setPreferableBackend(cv2.dnn.DNN_BACKEND_OPENCV)
    net.setPreferableTarget(cv2.dnn.DNN_TARGET_CPU)

    # Цвета для разных классов
    colors = np.random.uniform(0, 255, size=(len(classes), 3))

    print("Начинаем детекцию...")
    print("Управление:")
    print("  - Нажмите 'q' для выхода")
    print("  - Нажмите 'p' для паузы")

    frame_count = 0
    total_vehicles = 0
    paused = False

    # Настройки для оптимизации частоты обработки
    PROCESS_EVERY_N_FRAMES = 5  # Обрабатываем каждый 5-й кадр
    RESIZE_FACTOR = 0.6  # Уменьшаем размер кадра для обработки

    while True:
        if not paused:
            ret, frame = cap.read()
            if not ret:
                print("Конец видео или ошибка чтения")
                break

            frame_count += 1

            # Пропускаем кадры для оптимизации производительности
            if frame_count % PROCESS_EVERY_N_FRAMES != 0:
                # Показываем кадр без обработки
                cv2.imshow('YOLO Vehicle Detection - Press Q to quit', frame)
                if cv2.waitKey(1) & 0xFF == ord('q'):
                    break
                continue

            # Уменьшаем размер кадра для обработки
            if RESIZE_FACTOR < 1.0:
                original_height, original_width = frame.shape[:2]
                new_width = int(original_width * RESIZE_FACTOR)
                new_height = int(original_height * RESIZE_FACTOR)
                processing_frame = cv2.resize(frame, (new_width, new_height))
            else:
                processing_frame = frame.copy()
                new_width, new_height = original_width, original_height

            # Подготовка изображения для YOLO
            blob = cv2.dnn.blobFromImage(processing_frame, 1 / 255.0, (416, 416),
                                         swapRB=True, crop=False)
            net.setInput(blob)

            # Детекция
            outputs = net.forward(output_layers)

            # Обработка результатов
            boxes = []
            confidences = []
            class_ids = []

            for output in outputs:
                for detection in output:
                    scores = detection[5:]
                    class_id = np.argmax(scores)
                    confidence = scores[class_id]

                    # Фильтруем только транспортные средства с уверенностью > 50%
                    if confidence > 0.50 and class_id in [2, 5, 7]:  # 2:car, 5:bus, 7:truck
                        center_x = int(detection[0] * new_width)
                        center_y = int(detection[1] * new_height)
                        w = int(detection[2] * new_width)
                        h = int(detection[3] * new_height)

                        x = int(center_x - w / 2)
                        y = int(center_y - h / 2)

                        # Масштабируем координаты обратно к исходному размеру
                        if RESIZE_FACTOR < 1.0:
                            scale_x = original_width / new_width
                            scale_y = original_height / new_height
                            x = int(x * scale_x)
                            y = int(y * scale_y)
                            w = int(w * scale_x)
                            h = int(h * scale_y)

                        boxes.append([x, y, w, h])
                        confidences.append(float(confidence))
                        class_ids.append(class_id)

            # Применяем Non-Maximum Suppression для удаления дубликатов
            indexes = cv2.dnn.NMSBoxes(boxes, confidences, 0.5, 0.4)

            # Рисуем bounding boxes
            vehicles_detected = 0

            if len(boxes) > 0:
                if len(indexes) > 0:
                    for i in indexes.flatten():
                        vehicles_detected += 1
                        x, y, w, h = boxes[i]
                        color = colors[class_ids[i]]

                        # Рисуем прямоугольник
                        cv2.rectangle(frame, (x, y), (x + w, y + h), color, 2)

                        # Добавляем подпись
                        label = f"{classes[class_ids[i]]} {confidences[i]:.2f}"
                        cv2.putText(frame, label, (x, y - 10),
                                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

            total_vehicles += vehicles_detected

            # Добавляем информацию на кадр
            info_text = f"Vehicles: {vehicles_detected} | Frame: {frame_count}"
            cv2.putText(frame, info_text, (10, 30),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 255), 2)

            # Показываем статистику каждые 30 кадров
            if frame_count % 30 == 0:
                print(f"Кадр {frame_count}: обнаружено {vehicles_detected} транспортных средств")

        # Показываем кадр
        cv2.imshow('YOLO Vehicle Detection - Press Q to quit', frame)

        # Обработка клавиш
        key = cv2.waitKey(1) & 0xFF
        if key == ord('q'):
            break
        elif key == ord('p'):
            paused = not paused

    cap.release()
    cv2.destroyAllWindows()
    print(f"Детекция завершена!")
    print(f"Обработано кадров: {frame_count}")


if __name__ == "__main__":
    detect_cars_yolo()
